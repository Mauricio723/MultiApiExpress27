 static async crearTablaImagenes() {
        const string_query = "CREATE TABLE galeria_pinturas (id INT AUTO_INCREMENT PRIMARY KEY, " 
                + "titulo_img VARCHAR(100) NOT NULL, " 
                + "url_imagen VARCHAR(250) NOT NULL, " 
                + "orden_imagen INT, "
                + "id_pintor INT NOT NULL, FOREIGN KEY (id_pintor) REFERENCES galeria_pintores(id));";

        const retornoAlCrearTablaImg = await dbconnection.query(string_query); 

        return retornoAlCrearTablaImg;
    }

static async agregarPintor(lista_pintores) {

        let stringValuesPintores = "";

        let esPrimerPintor = true;

        lista_pintores.forEach(pintor => {
        
            if (pintor.nombre_pintor !== "pintor sin definir") {
                if(esPrimerPintor) {
                    stringValuesPintores += "('" + pintor.nombre_pintor + "')";
                    esPrimerPintor = false;

                } else {
                    stringValuesPintores += ", ('" + pintor.nombre_pintor + "')";
                }
                 
            }

        });

        const string_query = "INSERT INTO galeria_pintores (nombre_pintor)" 
            + " VALUES " + stringValuesPintores + ";"; 

        console.log("string con values: ", string_query); 
            
        const retornoAgregandoPintor = await dbconnection.query(string_query); 

        return retornoAgregandoPintor;      

        //return string_query;
    }

                    titulo_img: dato_imagen.titulo_img,
                    url_imagen: dato_imagen.url_imagen,
